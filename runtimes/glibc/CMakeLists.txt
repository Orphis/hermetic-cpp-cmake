# Builds the pieces needed to link against a given glibc version without a
# distribution sysroot, following hermeticbuild/hermetic-llvm
# (3rd_party/libc/glibc, runtimes/glibc), which in turn follows Zig:
#   * Scrt1.o / crt1.o and libc_nonshared.a compiled from the glibc sources,
#   * stub shared libraries (libc.so.6, libm.so.6, ...) generated from Zig's
#     abilists blob, exporting exactly the versioned symbols of that glibc,
#   * linker scripts (libc.so, libm.so, ...) and empty crti.o / crtn.o,
#   * the glibc headers and Linux UAPI headers installed to usr/include.
#
# Inputs:
#   GLIBC_SOURCE_DIR, GLIBC_VERSION (2.28), GLIBC_ARCH (x86_64|aarch64|riscv64|s390x|armv7)
#   GLIBC_HEADERS_DIR (…/<triple>/include), KERNEL_HEADERS_DIR (…/<arch>/include)
#   GLIBC_STUBS_GENERATOR (host executable), GLIBC_ABILISTS (file)
#   CMAKE_INSTALL_PREFIX (runtime set root)

cmake_minimum_required(VERSION 3.19)
project(hermetic_llvm_glibc C ASM)

foreach(v GLIBC_SOURCE_DIR GLIBC_VERSION GLIBC_ARCH GLIBC_HEADERS_DIR KERNEL_HEADERS_DIR GLIBC_STUBS_GENERATOR GLIBC_ABILISTS)
  if(NOT ${v})
    message(FATAL_ERROR "${v} is required")
  endif()
endforeach()
set(src "${GLIBC_SOURCE_DIR}")
set(overrides "${CMAKE_CURRENT_SOURCE_DIR}/overrides")
set(gen "${CMAKE_BINARY_DIR}/generated")
set(triple "${CMAKE_C_COMPILER_TARGET}")

# ---- Files normally produced by glibc's configure -----------------------------
file(WRITE "${gen}/config.h" "// Empty: every constant is passed as a define.\n")
file(WRITE "${gen}/abi-tag.h" "#define __ABI_TAG_OS 0\n#ifndef __ABI_TAG_VERSION\n# define __ABI_TAG_VERSION 2,0,0\n#endif\n")
set(modules iconvprogs iconvdata ldconfig libmemusage libpcprofile librpcsvc locale_programs memusagestat
  nonlib nscd extramodules libnldbl libsupport testsuite testsuite_internal LIBS_BEGIN rtld libc
  libBrokenLocale libpthread libthread_db libcrypt libdl libgcc_s libnsl libc_malloc_debug libutil
  libnss_ldap libnss_dns libnss_compat libmvec libresolv libnss_db libm libnss_files librt libnss_hesiod libanl)
set(content "/* AUTOGENERATED BY gen-libc-modules.awk, DO NOT EDIT.  */\n\n")
set(i 1)
foreach(m IN LISTS modules)
  string(APPEND content "#define MODULE_${m} ${i}\n")
  math(EXPR i "${i} + 1")
endforeach()
file(WRITE "${gen}/include/libc-modules.h" "${content}")

# ---- Include search path (port of helpers.bzl glibc_includes) ---------------
set(cpu "${GLIBC_ARCH}")
set(os_abi_variant "")
set(abi_variant "")
set(arch_parent "")
set(wordsize wordsize-64)
if(cpu STREQUAL "x86_64")
  set(os_abi_variant "sysdeps/unix/sysv/linux/x86_64/64")
  set(arch_parent "sysdeps/unix/sysv/linux/x86" "sysdeps/x86")
elseif(cpu STREQUAL "riscv64")
  set(os_abi_variant "sysdeps/unix/sysv/linux/riscv/rv64")
  set(cpu riscv)
elseif(cpu STREQUAL "s390x")
  set(os_abi_variant "sysdeps/unix/sysv/linux/s390/s390-64")
  set(abi_variant "sysdeps/s390/s390-64")
  set(cpu s390)
elseif(cpu STREQUAL "armv7")
  set(wordsize wordsize-32)
  set(cpu arm)
endif()
set(glibc_includes "${gen}" "${gen}/include" "${src}/include" ${os_abi_variant} ${abi_variant}
  "sysdeps/unix/sysv/linux/${cpu}" ${arch_parent}
  "sysdeps/${cpu}/nptl" "sysdeps/unix/sysv/linux/generic" "sysdeps/unix/sysv/linux/include"
  "sysdeps/unix/sysv/linux" "sysdeps/nptl" "sysdeps/pthread" "sysdeps/unix/sysv" "sysdeps/unix/${cpu}"
  "sysdeps/unix" "sysdeps/${cpu}" "sysdeps/${wordsize}" "sysdeps/generic" ".")
set(abs_includes "")
foreach(d IN LISTS glibc_includes)
  if(IS_ABSOLUTE "${d}")
    list(APPEND abs_includes "${d}")
  else()
    list(APPEND abs_includes "${src}/${d}")
  endif()
endforeach()
set(system_includes "${GLIBC_HEADERS_DIR}" "${KERNEL_HEADERS_DIR}")

set(CMAKE_C_FLAGS_RELEASE "")
set(CMAKE_ASM_FLAGS_RELEASE "")
set(common_flags -nostdlibinc -nobuiltininc -Wno-nonportable-include-path -Wa,--noexecstack -Qunused-arguments)

# ---- Scrt1.o: abi-note + init + start, combined into one object ---------------
add_library(glibc_abi_note OBJECT "${overrides}/abi-note-2.31.S")
target_include_directories(glibc_abi_note PRIVATE "${src}/csu" ${abs_includes})
target_compile_options(glibc_abi_note PRIVATE ${common_flags})
target_compile_definitions(glibc_abi_note PRIVATE _LIBC_REENTRANT MODULE_NAME=libc TOP_NAMESPACE=glibc ASSEMBLER)

add_library(glibc_init OBJECT "${src}/csu/init.c")
target_compile_options(glibc_init PRIVATE ${common_flags} -fno-addrsig)

if(GLIBC_ARCH STREQUAL "riscv64")
  set(start_s "${src}/sysdeps/riscv/start.S")
elseif(GLIBC_ARCH STREQUAL "s390x")
  set(start_s "${src}/sysdeps/s390/s390-64/start.S")
elseif(GLIBC_ARCH STREQUAL "armv7")
  set(start_s "${src}/sysdeps/arm/start.S")
else()
  set(start_s "${src}/sysdeps/${GLIBC_ARCH}/start.S")
endif()
add_library(glibc_start OBJECT "${start_s}")
target_include_directories(glibc_start PRIVATE ${abs_includes})
target_include_directories(glibc_start SYSTEM PRIVATE ${system_includes})
target_compile_options(glibc_start PRIVATE ${common_flags}
  "--include=${gen}/include/libc-modules.h" "--include=${src}/include/libc-symbols.h")
target_compile_definitions(glibc_start PRIVATE _LIBC_REENTRANT MODULE_NAME=libc PIC SHARED TOP_NAMESPACE=glibc ASSEMBLER)

set(lib "${CMAKE_BINARY_DIR}/lib")
file(MAKE_DIRECTORY "${lib}")
add_custom_command(OUTPUT "${lib}/Scrt1.o"
  COMMAND "${CMAKE_C_COMPILER}" --target=${triple} -fuse-ld=lld -nostdlib -r
          -o "${lib}/Scrt1.o" $<TARGET_OBJECTS:glibc_abi_note> $<TARGET_OBJECTS:glibc_init> $<TARGET_OBJECTS:glibc_start>
  DEPENDS glibc_abi_note glibc_init glibc_start COMMAND_EXPAND_LISTS)
# glibc builds crt1.o from the same sources with SHARED/PIC (see upstream TODO).
add_custom_command(OUTPUT "${lib}/crt1.o" COMMAND ${CMAKE_COMMAND} -E copy "${lib}/Scrt1.o" "${lib}/crt1.o" DEPENDS "${lib}/Scrt1.o")

# ---- crti.o / crtn.o: empty ------------------------------------------------------
file(WRITE "${gen}/empty.c" "")
foreach(t crti crtn)
  add_library(${t} OBJECT "${gen}/empty.c")
  target_compile_options(${t} PRIVATE -fno-addrsig)
  add_custom_command(OUTPUT "${lib}/${t}.o"
    COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_OBJECTS:${t}>" "${lib}/${t}.o" DEPENDS ${t} COMMAND_EXPAND_LISTS)
endforeach()

# ---- libc_nonshared.a ---------------------------------------------------------------
set(nonshared_sources
  "${overrides}/atexit.c" "${overrides}/at_quick_exit.c" "${overrides}/pthread_atfork.c"
  "${src}/debug/stack_chk_fail_local.c")
if(GLIBC_VERSION VERSION_LESS 2.33)
  # stat functions were redirected to xstat until 2.33.
  list(APPEND nonshared_sources
    "${src}/io/stat.c" "${src}/io/fstat.c" "${src}/io/lstat.c" "${src}/io/stat64.c" "${src}/io/fstat64.c"
    "${src}/io/lstat64.c" "${src}/io/fstatat.c" "${src}/io/fstatat64.c" "${src}/io/mknodat.c" "${src}/io/mknod.c")
endif()
if(GLIBC_VERSION VERSION_LESS 2.34)
  # __libc_start_main required statically linked init/fini callbacks until 2.34.
  list(APPEND nonshared_sources "${overrides}/elf-init-2.31.c")
endif()
add_library(c_nonshared STATIC ${nonshared_sources})
set_target_properties(c_nonshared PROPERTIES OUTPUT_NAME c_nonshared ARCHIVE_OUTPUT_DIRECTORY "${lib}")
target_include_directories(c_nonshared PRIVATE "${src}/csu" ${abs_includes})
target_include_directories(c_nonshared SYSTEM PRIVATE ${system_includes})
target_compile_options(c_nonshared PRIVATE
  -nostdlibinc -std=gnu11 -fgnu89-inline -fmerge-all-constants -frounding-math
  -Wno-unsupported-floating-point-opt -fno-common -fmath-errno -fPIC -ftls-model=initial-exec
  -Wno-ignored-attributes -Qunused-arguments -Wno-nonportable-include-path
  "--include=${gen}/include/libc-modules.h" "--include=${src}/include/libc-symbols.h" -O2)
target_compile_definitions(c_nonshared PRIVATE NO_INITFINI _LIBC_REENTRANT MODULE_NAME=libc PIC LIBC_NONSHARED=1 TOP_NAMESPACE=glibc)
if(GLIBC_ARCH STREQUAL "x86_64")
  target_compile_definitions(c_nonshared PRIVATE CAN_USE_REGISTER_ASM_EBP)
endif()

# ---- Stub shared libraries from the abilists blob ----------------------------------
if(GLIBC_ARCH STREQUAL "armv7")
  set(zig_target "arm-linux-gnueabihf.${GLIBC_VERSION}")
else()
  set(zig_target "${GLIBC_ARCH}-linux-gnu.${GLIBC_VERSION}")
endif()
set(stubs "${CMAKE_BINARY_DIR}/stubs")
set(stub_libs c dl ld m pthread resolv rt util)
set(stub_asm "")
foreach(l IN LISTS stub_libs)
  list(APPEND stub_asm "${stubs}/${l}.s")
endforeach()
add_custom_command(OUTPUT ${stub_asm} "${stubs}/all.map"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${stubs}"
  COMMAND "${GLIBC_STUBS_GENERATOR}" -target "${zig_target}" -o "${stubs}" "${GLIBC_ABILISTS}"
  DEPENDS "${GLIBC_ABILISTS}" COMMENT "Generating glibc ${GLIBC_VERSION} stubs for ${zig_target}")

set(so_versions_ld 2)
set(so_versions_c 6)
set(so_versions_dl 2)
set(so_versions_m 6)
set(so_versions_pthread 0)
set(so_versions_resolv 2)
set(so_versions_rt 1)
set(so_versions_util 1)
if(GLIBC_ARCH STREQUAL "x86_64")
  set(dynamic_linker ld-linux-x86-64.so.2)
elseif(GLIBC_ARCH STREQUAL "aarch64")
  set(dynamic_linker ld-linux-aarch64.so.1)
elseif(GLIBC_ARCH STREQUAL "armv7")
  set(dynamic_linker ld-linux-armhf.so.3)
elseif(GLIBC_ARCH STREQUAL "riscv64")
  set(dynamic_linker ld-linux-riscv64-lp64d.so.1)
elseif(GLIBC_ARCH STREQUAL "s390x")
  set(dynamic_linker ld64.so.1)
else()
  message(FATAL_ERROR "No dynamic linker name for ${GLIBC_ARCH}")
endif()

set(stub_outputs "")
foreach(l IN LISTS stub_libs)
  if(l STREQUAL "ld")
    set(soname "${dynamic_linker}")
  else()
    set(soname "lib${l}.so.${so_versions_${l}}")
  endif()
  set(extra "")
  if(l STREQUAL "c")
    # _IO_stdin_used lives in .rodata and is relocated at runtime.
    set(extra -Wl,-z,notext)
  endif()
  add_custom_command(OUTPUT "${lib}/${soname}"
    COMMAND "${CMAKE_C_COMPILER}" --target=${triple} -fuse-ld=lld -nostdlib -shared -Wno-unused-command-line-argument
            "-Wl,--version-script=${stubs}/all.map" "-Wl,-soname,${soname}" ${extra}
            -o "${lib}/${soname}" "${stubs}/${l}.s"
    DEPENDS "${stubs}/${l}.s" "${stubs}/all.map")
  list(APPEND stub_outputs "${lib}/${soname}")
  if(NOT l STREQUAL "ld")
    # Linker script with the name the linker looks for.
    if(l STREQUAL "c")
      file(WRITE "${lib}/lib${l}.so" "GROUP(\n${soname}\nlibc_nonshared.a\nAS_NEEDED(${dynamic_linker})\n)\n")
    else()
      file(WRITE "${lib}/lib${l}.so" "GROUP(\n${soname}\n)\n")
    endif()
    list(APPEND stub_outputs "${lib}/lib${l}.so")
  endif()
endforeach()

add_custom_target(glibc_runtime ALL DEPENDS "${lib}/Scrt1.o" "${lib}/crt1.o" "${lib}/crti.o" "${lib}/crtn.o" ${stub_outputs} c_nonshared)

# ---- Install -----------------------------------------------------------------------------
install(FILES "${lib}/Scrt1.o" "${lib}/crt1.o" "${lib}/crti.o" "${lib}/crtn.o" ${stub_outputs} DESTINATION usr/lib)
install(TARGETS c_nonshared ARCHIVE DESTINATION usr/lib)
install(DIRECTORY "${GLIBC_HEADERS_DIR}/" DESTINATION usr/include)
install(DIRECTORY "${KERNEL_HEADERS_DIR}/" DESTINATION usr/include)
